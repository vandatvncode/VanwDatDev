local HttpService = game:GetService("HttpService") -- Đường dẫn Firebase Database của bạn local FIREBASE_URL = "https://vd12-c514c-default-rtdb.asia-southeast1.firebasedatabase.app/keys/" local httpRequest = (syn and syn.request) or (http and http.request) or http_request or request local function verifyKeyWithFirebase(userKey) if not httpRequest then return false, "Executor không hỗ trợ HTTP Request" end local requestUrl = FIREBASE_URL .. userKey .. ".json" local success, response = pcall(function() return httpRequest({ Url = requestUrl, Method = "GET", Headers = { ["Content-Type"] = "application/json" } }) end) if not success or not response then return false, "Không thể kết nối tới Firebase" end if response.Body == "null" or response.Body == "" then return false, "Key không tồn tại!" end local ok, keyData = pcall(function() return HttpService:JSONDecode(response.Body) end) if not ok or not keyData then return false, "Lỗi đọc dữ liệu từ Firebase" end local now = os.time() if keyData.expires_at and now > keyData.expires_at then return false, "Key đã hết hạn sử dụng!" end return true, "Key hợp lệ!" end
